rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Businesses / Users
    match /users/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
      // New fields: preferredFinanceAgentGst, roles, isAlsoTrader
    }
    
    // Orders
    match /orders/{orderId} {
      allow read: if request.auth != null && (
        resource.data.buyerId == request.auth.uid || 
        resource.data.sellerId == request.auth.uid || 
        resource.data.financialAgentId == request.auth.uid || 
        // Allow finance agent to see orders assigned to them by GST
        (resource.data.financeAgentGst != null && resource.data.financeAgentGst == get(/databases/$(database)/documents/users/$(request.auth.uid)).data.gstNumber)
      );
      
      allow create: if request.auth != null;
      
      allow update: if request.auth != null && (
        resource.data.buyerId == request.auth.uid || 
        resource.data.sellerId == request.auth.uid ||
        // Allow finance agent to update status
        (resource.data.financeAgentGst != null && resource.data.financeAgentGst == get(/databases/$(database)/documents/users/$(request.auth.uid)).data.gstNumber)
      );
    }
  }
}
