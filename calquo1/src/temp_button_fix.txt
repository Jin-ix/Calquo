               <Button 
                  type="button"
                  onClick={() => {
                     if (!user) {
                       toast.error('Please login to send purchase request');
                       return;
                     }

                     if (totalSelectedQty === 0) {
                       toast.error('Please select at least one item');
                       return;
                     }

                     setCheckoutPhase('checkout');
                  }}

               >
                  <CreditCard className="h-4 w-4 mr-2" />

               </Button>
